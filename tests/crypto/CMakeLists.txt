cmake_minimum_required(VERSION 3.22)
project(adiantum_host C)
set(CMAKE_C_STANDARD 11)
add_subdirectory(../../Crypto crypto)
add_library(adiantum_test SHARED exports.c)
target_link_libraries(adiantum_test PRIVATE adiantum)

if(NOT MSVC)
    set_target_properties(adiantum PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_options(adiantum PRIVATE -Wall -Wextra -Wpedantic)
endif()
